local count, seen, dice, a, b, rand
global prize

fn GAME (count)
seen[1] = FALSE
seen[2] = FALSE
seen[3] = FALSE
seen[4] = FALSE
seen[5] = FALSE
seen[6] = FALSE
call 4 + 5 (count, ref seen)
prize = seen[6]
return

fn ROLL_MULT (count, seen)
if count <= 0 return
call ROLL_ONCE (ref dice)
seen[dice] = TRUE
count = count - 1
repeat

fn ROLL_ONCE (dice)
a = rand[0] <= 0
b = (rand[1] <= 0) + (rand[2] <= 0)
dice = 1 + a + b + b
return
